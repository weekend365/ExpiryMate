async function contrast(page,texts){
 const style=await page.addStyleTag({content:'[data-qa-text]{color:transparent!important;-webkit-text-fill-color:transparent!important;text-shadow:none!important}*{caret-color:transparent!important}'});
 const background=PNG.sync.read(await page.screenshot({animations:'disabled'}));
 await style.evaluate(e=>e.textContent='[data-qa-text]{color:black!important;-webkit-text-fill-color:black!important;text-shadow:none!important}*{caret-color:transparent!important}');
 const black=PNG.sync.read(await page.screenshot({animations:'disabled'}));
 await style.evaluate(e=>e.textContent='[data-qa-text]{color:white!important;-webkit-text-fill-color:white!important;text-shadow:none!important}*{caret-color:transparent!important}');
 const white=PNG.sync.read(await page.screenshot({animations:'disabled'}));
 const idColor=id=>[8+(id%16)*16,8+(Math.floor(id/16)%16)*16,104];
 await style.evaluate((e,colors)=>{e.textContent=colors.map(({id,c})=>'[data-qa-text=\"'+id+'\"]{color:rgb('+c.join(',')+')!important;-webkit-text-fill-color:rgb('+c.join(',')+')!important;text-shadow:none!important}').join('')+'*{caret-color:transparent!important}';},texts.map(t=>({id:t.id,c:idColor(t.id)})));
 const identity=PNG.sync.read(await page.screenshot({animations:'disabled'}));await style.evaluate(e=>e.remove());
 const results=[];
 for(const t of texts){
  if(t.disabled||t.width<=0||t.height<=0)continue;const channels=t.color.match(/[\d.]+/g)?.map(Number);if(!channels||channels.length<3)continue;
  const foreground=lum(channels.slice(0,3)),identityColor=idColor(t.id);let min=100,pixels=0;
  for(let y=Math.max(0,Math.ceil(t.y));y<Math.min(background.height,Math.floor(t.y+t.height));y++)for(let x=Math.max(0,Math.ceil(t.x));x<Math.min(background.width,Math.floor(t.x+t.width));x++){
   const index=(y*background.width+x)*4;
   if(white.data[index]-black.data[index]<200||white.data[index+1]-black.data[index+1]<200||white.data[index+2]-black.data[index+2]<200)continue;
   if(identityColor.some((c,k)=>Math.abs(identity.data[index+k]-c)>2))continue;
   const b=lum([background.data[index],background.data[index+1],background.data[index+2]]);min=Math.min(min,(Math.max(foreground,b)+.05)/(Math.min(foreground,b)+.05));pixels++;
  }
  if(pixels)results.push({id:t.id,text:t.text,color:t.color,fontSize:t.fontSize,pixels,minContrast:min,passes:min>=4.5});
 }
 return results;
}
