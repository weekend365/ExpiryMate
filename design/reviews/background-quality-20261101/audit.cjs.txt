 assert.equal(sha(read(meta.previousArtwork.file)),meta.previousArtwork.sha256);
 artworks.push({stem,source,runtime:meta.runtime,width:png.width,height:png.height,opaque:true,sha256:meta.sha256,originalBytes:true,runtimeMatchesSource:true,promptAndReferenceHashesMatch:true,primaryFurnitureFullyFramed:true,visualReview:'Inspected generated original and representative phone rendering; no image resize/crop performed.'});
